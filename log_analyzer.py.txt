from collections import Counter

LOG_FILE = "../logs/sample.log"

keywords = {
    "ERROR": "High",
    "WARNING": "Medium",
    "Failed login": "Medium",
    "Unauthorized": "High"
}

incident_counter = Counter()

print("=" * 50)
print("Security Log Analysis")
print("=" * 50)

with open(LOG_FILE, "r") as logfile:
    for line in logfile:
        for keyword, severity in keywords.items():
            if keyword in line:
                incident_counter[keyword] += 1
                print(f"[{severity}] {line.strip()}")

print("\nSummary")
print("-" * 30)

for incident, count in incident_counter.items():
    print(f"{incident}: {count}")

print("\nAnalysis Complete.")
